library(tidyverse)

DATA_FILE <- "fmr2027.csv"  # relative to this .qmd file
REF_BEDROOMS <- 2           # bedroom count for single-size views and ratios
N_EXTREME <- 12             # areas shown at each end of the extremes plot
JITTER_SEED <- 1            # deterministic jitter
fmr_raw <- read_csv(
  DATA_FILE,
  col_types = cols(
    fips = col_character(), stusps = col_character(),
    state = col_character(), countyname = col_character(),
    county_town_name = col_character(), hud_area_code = col_character(),
    hud_area_name = col_character(), metro = col_integer(),
    pop2024 = col_double(), bedrooms = col_integer(), fmr = col_double()
  )
)

# fail loudly if the structure we rely on is not there
stopifnot(
  nrow(problems(fmr_raw)) == 0,
  all(fmr_raw$metro %in% c(0L, 1L)),
  REF_BEDROOMS %in% fmr_raw$bedrooms
)
# Records: one row per geographic record and bedroom count.
fmr <- fmr_raw |>
  mutate(
    metro = factor(metro, levels = c(0, 1), labels = c("nonmetro", "metro")),
    # fips = state (2) + county (3) + county subdivision (5);
    # a 99999 subdivision code means the whole county
    record_type = if_else(str_ends(fips, "99999"), "county", "sub-county")
  )

# Areas: one row per HUD FMR area and bedroom count. FMR must be identical
# for every record in an area; population is summed over the records.
stopifnot(
  fmr |>
    summarise(k = n_distinct(fmr), .by = c(hud_area_code, bedrooms)) |>
    pull(k) |>
    (\(k) all(k == 1))()
)

areas <- fmr |>
  summarise(
    fmr = first(fmr),
    pop = sum(pop2024),
    n_records = n(),
    .by = c(hud_area_code, hud_area_name, metro, bedrooms)
  )

# An area that crosses a state line is kept once in each state it touches.
area_states <- fmr |>
  distinct(hud_area_code, stusps, metro, bedrooms, fmr)

weighted_ecdf <- function(fmr, weight, unit) {
  o <- order(fmr)
  weight <- rep_len(weight, length(fmr))
  tibble(
    bedrooms = REF_BEDROOMS,
    fmr = fmr[o],
    cumulative = cumsum(weight[o]) / sum(weight),
    unit = unit
  )
}

Distribution of fmr and bedroom structure

quartile_summary <- function(data, label) {
  data |>
    summarise(
      n = n(), min = min(fmr),
      q1 = quantile(fmr, 0.25), median = median(fmr),
      q3 = quantile(fmr, 0.75), max = max(fmr), mean = mean(fmr),
      # Bowley skewness: > 0 means a longer upper half of the box
      bowley = ((q3 - median) - (median - q1)) / (q3 - q1),
      .by = bedrooms
    ) |>
    mutate(unit = label, .before = 1) |>
    arrange(bedrooms)
}

weighted_median <- function(x, w) {
  if (sum(w) == 0) return(NA_real_)
  o <- order(x)
  x[o][which(cumsum(w[o]) >= sum(w) / 2)[1]]
}
